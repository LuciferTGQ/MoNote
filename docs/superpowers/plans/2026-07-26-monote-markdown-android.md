# MoNote Markdown Android MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a side-loadable, fully offline Android 11+ APK named 墨笺 (MoNote) that imports Markdown from WeChat or the system, manages a public document library, edits safely, and renders GFM, math, Mermaid, footnotes, code, and local images.

**Architecture:** A Kotlin/Jetpack Compose shell owns permissions, storage, indexing, navigation, and native file integration. One trusted local WebView document surface owns CodeMirror editing and Markdown rendering; a typed message bridge connects it to a `DocumentSession` that handles recovery, atomic saves, and conflicts. Real content lives under `Documents/MoNote/`; Room is an index and `_MoNoteSystem/catalog.json` is the recoverable metadata mirror.

**Tech Stack:** AGP 8.13.2, Gradle 8.13, JDK 17, Kotlin 2.3.21, compile/target SDK 36, min SDK 30, Compose BOM 2026.06.00, Room 2.8.4, WebKit 1.16.0, DataStore 1.2.1, Navigation Compose 2.9.8, Kotlin serialization 1.11.0, Node 22, Vite 8.1.5, Vitest 4.1.10, CodeMirror 6, Markdown-it 14.3.0, KaTeX 0.18.1, Mermaid 11.16.0, Highlight.js 11.11.1, DOMPurify 3.4.12.

        val system = root.resolve("_MoNoteSystem")
        return LibraryDirectories(
            root = root.apply(File::mkdirs),
            inbox = root.resolve("收件箱").apply(File::mkdirs),
            system = system.apply(File::mkdirs),
            recovery = system.resolve("recovery").apply(File::mkdirs),
            backups = system.resolve("backups").apply(File::mkdirs),
            trash = system.resolve("trash").apply(File::mkdirs),
        )

class SafePathPolicy(private val root: File) {
    private val canonicalRoot = root.canonicalFile

    fun resolve(relative: String): File {
        val candidate = canonicalRoot.resolve(relative).canonicalFile
        if (candidate != canonicalRoot && !candidate.toPath().startsWith(canonicalRoot.toPath())) {
            throw UnsafePathException(relative)
        }
        return candidate
    }

    fun sanitizeFileName(raw: String): String =
        raw.replace(Regex("""[\\/:*?"<>|]"""), "_").trim().ifBlank { "未命名.md" }
}

- [ ] **Step 5: Implement write-temp, flush, verify, replace, and fingerprint**

`AtomicTextStore.replace` writes UTF-8 to `<name>.monote-tmp`, flushes `FileDescriptor.sync()`, verifies SHA-256, preserves one backup, then replaces within the same directory. `DocumentFingerprint.from(file)` returns size, modified time, and SHA-256.

```kotlin
data class DocumentFingerprint(val size: Long, val modifiedAt: Long, val sha256: String) {
    companion object {
        fun sha256(bytes: ByteArray): String =
            MessageDigest.getInstance("SHA-256").digest(bytes).joinToString("") { "%02x".format(it) }

        fun sha256(text: String): String = sha256(text.toByteArray(Charsets.UTF_8))

        fun from(file: File): DocumentFingerprint =
            DocumentFingerprint(file.length(), file.lastModified(), sha256(file.readBytes()))
    }
}

class AtomicTextStore {
    fun replace(
        target: File,
        text: String,
        backup: File? = null,
        beforeCommit: () -> Unit = {},
    ) {
        val temp = File(target.parentFile, "${target.name}.monote-tmp")
        try {
            FileOutputStream(temp).use { out ->
                out.write(text.toByteArray(Charsets.UTF_8))
                out.fd.sync()
            }
            beforeCommit()
            check(DocumentFingerprint.sha256(temp.readBytes()) == DocumentFingerprint.sha256(text))
            if (target.exists() && backup != null) {
                backup.parentFile.mkdirs()
                Files.copy(target.toPath(), backup.toPath(), StandardCopyOption.REPLACE_EXISTING)
            }
            runCatching {
                Files.move(
                    temp.toPath(),
                    target.toPath(),
                    StandardCopyOption.REPLACE_EXISTING,
                    StandardCopyOption.ATOMIC_MOVE,
                )
            }.getOrElse {
                Files.move(temp.toPath(), target.toPath(), StandardCopyOption.REPLACE_EXISTING)
            }
        } finally {
            temp.delete()
        }
    }
}
```

- [ ] **Step 3: Implement source abstraction, decoding, validation, and copy commit**

```kotlin
interface DocumentSource {
    val displayName: String
    val mimeType: String?
    suspend fun open(): InputStream
}

enum class EncodingHint { UTF8, UTF16_LE, UTF16_BE }
data class DecodedText(val text: String, val encoding: String, val warnings: List<String> = emptyList())

class TextDecoder {
    fun decode(bytes: ByteArray, hint: EncodingHint? = null): DecodedText {
        val detected = hint ?: when {
            bytes.hasPrefix(0xEF, 0xBB, 0xBF) -> EncodingHint.UTF8
            bytes.hasPrefix(0xFF, 0xFE) -> EncodingHint.UTF16_LE
            bytes.hasPrefix(0xFE, 0xFF) -> EncodingHint.UTF16_BE
            else -> EncodingHint.UTF8
        }
        val bomLength = when {
            bytes.hasPrefix(0xEF, 0xBB, 0xBF) -> 3
            bytes.hasPrefix(0xFF, 0xFE) || bytes.hasPrefix(0xFE, 0xFF) -> 2
            else -> 0
        }
        val charset = when (detected) {
            EncodingHint.UTF8 -> Charsets.UTF_8
            EncodingHint.UTF16_LE -> Charsets.UTF_16LE
            EncodingHint.UTF16_BE -> Charsets.UTF_16BE
        }
        val decoder = charset.newDecoder()
            .onMalformedInput(CodingErrorAction.REPORT)
            .onUnmappableCharacter(CodingErrorAction.REPORT)
        val text = decoder.decode(ByteBuffer.wrap(bytes, bomLength, bytes.size - bomLength)).toString()
        return DecodedText(text, charset.name())
    }

    private fun ByteArray.hasPrefix(vararg values: Int): Boolean =
        size >= values.size && values.indices.all { this[it].toInt() and 0xFF == values[it] }
}

data class ImportedDocument(val file: File, val encoding: String, val warnings: List<String>)

class ImportCoordinator(
    private val pathPolicy: SafePathPolicy,
    private val decoder: TextDecoder,
) {
    suspend fun import(source: DocumentSource, destination: File): ImportedDocument = withContext(Dispatchers.IO) {
        val safe = pathPolicy.sanitizeFileName(source.displayName)
        require(safe.endsWith(".md", true) || safe.endsWith(".markdown", true))
        val bytes = source.open().use(InputStream::readBytes)
        val decoded = decoder.decode(bytes)
        val target = uniqueTarget(destination, safe)
        AtomicTextStore().replace(target, decoded.text)
        ImportedDocument(target, decoded.encoding, decoded.warnings)
    }

    private fun uniqueTarget(destination: File, safeName: String): File {
        val extension = safeName.substringAfterLast('.', missingDelimiterValue = "")
        val stem = safeName.removeSuffix(if (extension.isEmpty()) "" else ".$extension")
        var candidate = File(destination, safeName)
        var copy = 2
        while (candidate.exists()) {
            candidate = File(destination, "$stem ($copy).$extension")
            copy += 1
        }
        return candidate
    }
}
```
